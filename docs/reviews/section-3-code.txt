<?php

use Espo\Core\Container;
use Espo\Core\InjectableFactory;
use Espo\Core\Utils\Config;
use Espo\Core\Utils\Config\ConfigWriter;
use Espo\ORM\EntityManager;

class AfterInstall
{
    private const THEME = 'Bidar';
    private const LAYOUT = 'bidarLayout';

    public function run(Container $container): void
    {
        $config = $container->getByClass(Config::class);
        $configWriter = $container->getByClass(InjectableFactory::class)->create(ConfigWriter::class);

        $systemTheme = $config->get('theme');
        if (empty($systemTheme) || $systemTheme === 'Espo' || $systemTheme === 'Lucid') {
            $configWriter->set('theme', self::THEME);
        }

        if (!$config->get(self::LAYOUT)) {
            $configWriter->set(self::LAYOUT, 'split');
        }
    }
}

/**
 * Shared settings of the Bidar record view.
 */

/** Bottom panels of the Activity tab, below the quick actions. */
export const ACTIVITY_PANELS = ['stream'];

/**
 * Record view layouts: `split` puts the fields on the left and the activity on the right;
 * `classic` keeps fields and bottom panels on the left and side panels on the right, as EspoCRM.
 */
export const LAYOUTS = ['split', 'classic'];

/**
 * Whether the theme in use is a Bidar theme (Bidar, Bidar Monochrome, Bidar Brut): those declare
 * their style in the `bidarStyle` theme parameter.
 *
 * @param {import('view').default} view
 * @return {boolean}
 */
export function isBidarTheme(view) {
    return !!getStyle(view);
}

/**
 * The style of the theme in use (`bidar`, `monochrome`, `brut`), or null under another theme.
 *
 * @param {import('view').default} view
 * @return {string|null}
 */
export function getStyle(view) {
    const style = view.getThemeManager().getParam('bidarStyle');

    return typeof style === 'string' && style ? style : null;
}

/**
 * The record view layout: the user's choice (`bidarLayout` preference), else the administrator's
 * default (`bidarLayout` setting), else Split.
 *
 * @param {import('view').default} view
 * @return {'split'|'classic'}
 */
export function getLayout(view) {
    const chosen = view.getPreferences().get('bidarLayout');

    if (LAYOUTS.includes(chosen)) {
        return chosen;
    }

    const byDefault = view.getConfig().get('bidarLayout');

    return LAYOUTS.includes(byDefault) ? byDefault : LAYOUTS[0];
}

/**
 * @param {import('view').default} view
 * @return {boolean}
 */
export function isClassicLayout(view) {
    return getLayout(view) === 'classic';
}

/**
 * @param {string} tag
 * @param {string} [className]
 * @param {string} [text]
 * @return {HTMLElement}
 */
export function el(tag, className, text) {
    const element = document.createElement(tag);

    if (className) {
        element.className = className;
    }

    if (text !== undefined) {
        element.textContent = text;
    }

    return element;
}

/**
 * Administration > User Interface: adds the default record layout of the Bidar themes (Split or
 * Classic) right below the theme. It applies to the users who have not chosen their own in the
 * Appearance menu of the top bar. The style is the theme itself, chosen in the Theme field.
 *
 * The field is inserted into EspoCRM's own layout of the page instead of replacing it, so the page
 * keeps what a later EspoCRM version adds to it: in the free cell of the row below the theme, else
 * on a row of its own below it, else in a block of its own after the first one. If anything
 * fails, the page stays as EspoCRM made it.
 */

/** The settings field. */
const FIELD = 'bidarLayout';

export default class BidarUserInterfaceHandler {

    /**
     * @param {import('views/record/edit').default} view
     */
    constructor(view) {
        this.view = view;
    }

    process() {
        const view = /** @type {Object.<string, *>} */ this.view;

        if (view.layoutName !== 'userInterface') {
            return;
        }

        try {
            if (view.detailLayout) {
                // The layout was read before this handler ran (it was cached): the fields are
                // built again with the defaults, before the page is first shown.
                const layout = Espo.Utils.cloneDeep(view.detailLayout);

                if (!insertField(layout)) {
                    return;
                }

                view.detailLayout = layout;
                view.gridLayout = null;
                view.createMiddleView();

                return;
            }

            const modify = view.modifyDetailLayout.bind(view);

            view.hasModifyDetailLayout = true;

            view.modifyDetailLayout = layout => {
                modify(layout);
                insertField(layout);
            };
        } catch (e) {
            console.warn('Bidar: the appearance defaults could not be added to the page.', e);
        }
    }
}

/**
 * @param {Object[]} layout Panels of a record layout.
 * @return {boolean} Whether the field was added.
 */
function insertField(layout) {
    if (!Array.isArray(layout)) {
        return false;
    }

    const isCell = (cell, name) => !!cell && typeof cell === 'object' && cell.name === name;
    const hasCell = (row, name) => Array.isArray(row) && row.some(cell => isCell(cell, name));

    if (layout.some(panel => (panel.rows || []).some(row => hasCell(row, FIELD)))) {
        return false;
    }

    const cell = {name: FIELD};

    for (const panel of layout) {
        const rows = panel.rows || [];
        const index = rows.findIndex(row => hasCell(row, 'theme'));

        if (index === -1) {
            continue;
        }

        const next = rows[index + 1];

        // EspoCRM 10 leaves the first cell of that row free (Disable avatars is on the right)
        if (Array.isArray(next) && next.length === 2 && next[0] === false) {
            next[0] = cell;
        } else {
            rows.splice(index + 1, 0, [cell, false]);
        }

        return true;
    }

    layout.splice(Math.min(1, layout.length), 0, {rows: [[cell, false]]});

    return true;
}

/**
 * Appearance of the Bidar themes: style (Bidar, Monochrome, Brut: one per theme), palette (light,
 * dark, or the one of the operating system) and density (comfortable, compact).
 *
 * A plain script, loaded before the stylesheet, so that a page opens directly in the right style
 * and palette instead of flashing the default ones. It sets on the root element:
 * - data-bidar-style: the style, read by css/bidar-style-*.css;
 * - data-bidar-scheme="light|dark": the resolved palette, read by css/bidar-dark.css;
 * - data-bidar-density: the density, read by css/bidar-density.css.
 *
 * The style is the one of the theme: its stylesheet is css/bidar.css or css/bidar-<style>.css, and
 * EspoCRM swaps it for the user's own theme once it has loaded the preferences. The palette and the
 * density are the user's preferences; the browser remembers all three, so that they apply before
 * EspoCRM has loaded, login page included. The top bar button announces a new choice with the
 * `bidar:appearance` event. Under another theme the attributes have no effect.
 */
(() => {
    const MODES = ['light', 'dark', 'system'];
    const DENSITIES = ['comfortable', 'compact'];
    // Styles are open-ended: a value without its stylesheet simply leaves the default look.
    const STYLE_PATTERN = /^[a-z][a-z-]{0,30}$/;

    const KEYS = {
        mode: 'bidar-color-scheme',
        style: 'bidar-style',
        density: 'bidar-density',
    };

    const isValid = {
        mode: value => MODES.includes(value),
        style: value => typeof value === 'string' && STYLE_PATTERN.test(value),
        density: value => DENSITIES.includes(value),
    };

    const DEFAULTS = {
        mode: 'light',
        style: 'bidar',
        density: 'comfortable',
    };

    // The stylesheet of a Bidar theme: css/bidar.css (Bidar) or css/bidar-<style>.css.
    const STYLESHEET = /\/bidar-theme\/css\/bidar(?:-([a-z]+))?\.css(?:[?#]|$)/;

    const media = window.matchMedia ? window.matchMedia('(prefers-color-scheme: dark)') : null;

    const read = name => {
        try {
            const stored = localStorage.getItem(KEYS[name]);

            return isValid[name](stored) ? stored : DEFAULTS[name];
        } catch (e) {
            return DEFAULTS[name];
        }
    };

    const store = (name, value) => {
        try {
            localStorage.setItem(KEYS[name], value);
        } catch (e) {
            // Blocked storage: the choice still applies until the next page load.
        }
    };

    const state = {
        mode: read('mode'),
        style: read('style'),
        density: read('density'),
    };

    const apply = () => {
        const root = document.documentElement;
        const isDark = state.mode === 'dark' || (state.mode === 'system' && !!media && media.matches);

        root.dataset.bidarScheme = isDark ? 'dark' : 'light';
        root.dataset.bidarStyle = state.style;
        root.dataset.bidarDensity = state.density;
    };

    apply();

    // System: follows the operating system when it switches, at nightfall for instance.
    if (media && media.addEventListener) {
        media.addEventListener('change', () => {
            if (state.mode === 'system') {
                apply();
            }
        });
    }

    // The style follows the theme's stylesheet, also when EspoCRM swaps it for the user's theme.
    const followStylesheet = link => {
        const update = () => {
            const match = (link.getAttribute('href') || '').match(STYLESHEET);
            const style = match ? (match[1] || DEFAULTS.style) : null;

            if (style && isValid.style(style) && style !== state.style) {
                state.style = style;
                store('style', style);
                apply();
            }
        };

        update();

        new MutationObserver(update).observe(link, {attributes: true, attributeFilter: ['href']});
    };

    const stylesheet = document.getElementById('main-stylesheet');

    if (stylesheet) {
        followStylesheet(stylesheet);
    } else if (window.MutationObserver) {
        // The page declares the stylesheet right after the scripts.
        const observer = new MutationObserver(() => {
            const link = document.getElementById('main-stylesheet');

            if (link) {
                observer.disconnect();
                followStylesheet(link);
            }
        });

        observer.observe(document.documentElement, {childList: true, subtree: true});
    }

    document.addEventListener('bidar:appearance', event => {
        const detail = event.detail || {};

        Object.keys(KEYS).forEach(name => {
            if (name in detail && isValid[name](detail[name])) {
                state[name] = detail[name];
                store(name, detail[name]);
            }
        });

        apply();
    });
})();

/**
 * Collapsible groups in the side menu: a click on a title (CRM, Activities…) folds the items below
 * it, up to the next title. The state is remembered per user, in the browser.
 *
 * A plain script rather than a module: EspoCRM's navbar view has no setup handler, and this only
 * relies on the menu's markup. If that markup changes in another version, the menu stays as it is.
 */
(() => {
    const MENU_SELECTOR = '#navbar ul.tabs';
    // The stylesheet of a Bidar theme: css/bidar.css (Bidar) or css/bidar-<style>.css (Bidar
    // Monochrome, Bidar Brut…), the same rule as js/appearance.js.
    const THEME_STYLESHEET = /\/bidar-theme\/css\/bidar(?:-[a-z]+)?\.css(?:[?#]|$)/;
    // The user's own link, in the top right menu. The Last Viewed panel, also in the navbar, can
    // hold links to other users, whose stored state must not be read.
    const USER_SELECTOR = '#navbar .menu-container a[href^="#User/view/"]';
    const STORAGE_PREFIX = 'bidar-nav-groups-';

    let scheduled = false;
    let observed = null;

    const storageKey = () => {
        const link = document.querySelector(USER_SELECTOR);

        return STORAGE_PREFIX + (link ? link.getAttribute('href').split('/').pop() : 'default');
    };

    const load = () => {
        try {
            const stored = JSON.parse(localStorage.getItem(storageKey()));

            return Array.isArray(stored) ? stored : [];
        } catch (e) {
            return [];
        }
    };

    const save = collapsed => {
        try {
            localStorage.setItem(storageKey(), JSON.stringify(collapsed));
        } catch (e) {
            // Blocked storage: the group still folds until the next page load.
        }
    };

    const toggle = label => {
        const collapsed = load();

        save(collapsed.includes(label) ? collapsed.filter(item => item !== label) : [...collapsed, label]);
        update();
    };

    const decorate = (title, label, isCollapsed) => {
        let button = title.querySelector(':scope > div > .bidar-nav-toggle');

        if (!button) {
            const row = title.firstElementChild;
            const icon = document.createElement('span');

            button = document.createElement('button');
            button.type = 'button';
            button.className = 'bidar-nav-toggle';
            icon.className = 'fas fa-chevron-down';
            icon.setAttribute('aria-hidden', 'true');
            button.append(icon);
            row.append(button);
            row.addEventListener('click', () => toggle(label));
        }

        button.setAttribute('aria-label', label);
        button.setAttribute('aria-expanded', String(!isCollapsed));
        title.classList.toggle('bidar-nav-collapsed', isCollapsed);
    };

    function isBidarTheme() {
        return [...document.querySelectorAll('link[rel="stylesheet"]')]
            .some(link => THEME_STYLESHEET.test(link.getAttribute('href') || ''));
    }

    function update() {
        watch();

        const menu = document.querySelector(MENU_SELECTOR);

        if (!menu || !isBidarTheme()) {
            return;
        }

        const collapsed = load();
        let isFolding = false;

        [...menu.children].forEach(item => {
            if (item.classList.contains('tab-divider')) {
                const text = item.querySelector('.label-text');
                const label = text ? text.textContent.trim() : '';

                // A separator without text has no title to click: it just ends the group above.
                isFolding = !!label && collapsed.includes(label);

                if (label) {
                    decorate(item, label, isFolding);
                }

                return;
            }

            if (item.classList.contains('more')) {
                isFolding = false;
            }

            // The page in progress stays visible, so that the user always sees where they are.
            const isCurrent = item.classList.contains('active') || !!item.querySelector('.active');

            item.classList.toggle('bidar-nav-hidden', isFolding && !isCurrent);
        });
    }

    const schedule = () => {
        if (scheduled) {
            return;
        }

        scheduled = true;

        requestAnimationFrame(() => {
            scheduled = false;
            update();
        });
    };

    // EspoCRM redraws the menu and moves the current page marker without any event to listen to.
    const observer = new MutationObserver(schedule);

    // Only the header holds the menu: the rest of the page changes far more often. Until it exists
    // (and if it gets replaced), the whole document is watched.
    function watch() {
        const header = document.querySelector('#header');
        const target = header || document.documentElement;

        if (target !== observed) {
            observer.disconnect();
            observer.observe(target, {childList: true, subtree: true});
            observed = target;
        }
    }

    watch();
    window.addEventListener('hashchange', schedule);
})();
