/**
 * BidarCRM - KanbanColumn Component
 *
 * Renders an individual pipeline stage column with Didar-inspired chevron header,
 * aggregated financial metrics, quick-create trigger, and responsive drag-drop zone.
 */

import React, { useState } from 'react';
import { Plus } from 'lucide-react';
import { PipelineStage, Opportunity } from '../../../types/espocrm';
import { DealCard } from './DealCard';
import { formatToman, toPersianDigits } from '../../../lib/currency';

export interface KanbanColumnProps {
  stage: PipelineStage;
  deals: Opportunity[];
  onDropDeal: (dealId: string, stageId: string, stageName: string) => void;
  onSelectDeal?: (dealId: string) => void;
  onQuickAddDeal?: (stageId: string, stageName: string) => void;
  className?: string;
}

export const KanbanColumn: React.FC<KanbanColumnProps> = ({
  stage,
  deals,
  onDropDeal,
  onSelectDeal,
  onQuickAddDeal,
  className = '',
}) => {
  const [isDragOver, setIsDragOver] = useState(false);

  // Filter deals matching this stage by ID, Name, or mapped status (e.g., Prospecting, Qualification)
  const stageDeals = deals.filter(
    (d) =>
      d.pipelineStageId === stage.id ||
      d.stage === stage.name ||
      Boolean(stage.mappedStatus && d.stage === stage.mappedStatus)
  );

  const totalAmount = stageDeals.reduce((sum, d) => sum + (d.amount || 0), 0);
  const color = stage.color || '#10B981';

  const handleDragOver = (e: React.DragEvent<HTMLDivElement>) => {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
    if (!isDragOver) setIsDragOver(true);
  };

  const handleDragLeave = (e: React.DragEvent<HTMLDivElement>) => {
    if (!e.currentTarget.contains(e.relatedTarget as Node)) {
      setIsDragOver(false);
    }
  };

  const handleDrop = (e: React.DragEvent<HTMLDivElement>) => {
    e.preventDefault();
    setIsDragOver(false);
    const dealId = e.dataTransfer.getData('text/plain');
    if (dealId) {
      onDropDeal(dealId, stage.id, stage.name);
    }
  };

  return (
    <div
      data-testid={`kanban-column-${stage.id}`}
      onDragOver={handleDragOver}
      onDragLeave={handleDragLeave}
      onDrop={handleDrop}
      className={`w-72 sm:w-80 flex-shrink-0 flex flex-col bg-slate-200/50 dark:bg-slate-900/60 border rounded-2xl transition-colors duration-150 overflow-hidden ${
        isDragOver
          ? 'border-emerald-500 bg-emerald-500/10 shadow-lg shadow-emerald-500/10'
          : 'border-slate-200/90 dark:border-slate-800/90'
      } ${className}`}
      dir="rtl"
    >
      {/* Column Stage Header - Didar Stepper Style */}
      <div className="p-3.5 border-b border-slate-200 dark:border-slate-800 bg-white dark:bg-slate-900/90 shadow-2xs">
        <div className="flex items-center justify-between gap-2 mb-1.5">
          <div className="flex items-center gap-2">
            <span
              className="w-2.5 h-2.5 rounded-full flex-shrink-0 shadow-xs"
              style={{ backgroundColor: color }}
            />
            <h3 className="font-bold text-xs sm:text-sm text-slate-800 dark:text-slate-100 tracking-tight">
              {stage.name}
            </h3>
          </div>
          <span className="text-[11px] font-medium text-slate-600 dark:text-slate-400 bg-slate-100 dark:bg-slate-800 px-2 py-0.5 rounded-md">
            {toPersianDigits(stageDeals.length)} معامله
          </span>
        </div>

        <div className="flex items-center justify-between text-[11px] text-slate-500 dark:text-slate-400">
          <span>مجموع ارزش:</span>
          <span className="font-semibold text-slate-700 dark:text-slate-200 font-mono" dir="ltr">
            {formatToman(totalAmount)}
          </span>
        </div>
      </div>

      {/* Quick Add Deal in Stage Box */}
      {onQuickAddDeal && (
        <div className="px-3 pt-3">
          <button
            type="button"
            onClick={() => onQuickAddDeal(stage.id, stage.name)}
            className="w-full flex items-center justify-center gap-1.5 py-2 px-3 border border-dashed border-emerald-500/50 dark:border-emerald-500/40 hover:border-emerald-600 dark:hover:border-emerald-400 bg-white/70 dark:bg-slate-800/50 hover:bg-emerald-50/50 dark:hover:bg-emerald-950/20 text-emerald-700 dark:text-emerald-400 text-xs font-medium rounded-xl transition cursor-pointer"
          >
            <Plus className="w-3.5 h-3.5" />
            <span>افزودن سریع معامله</span>
          </button>
        </div>
      )}

      {/* Scrollable Deal Cards Area */}
      <div className="flex-1 overflow-y-auto p-3 space-y-2.5 min-h-[140px]">
        {stageDeals.length === 0 ? (
          <div className="h-28 flex flex-col items-center justify-center border-2 border-dashed border-slate-200 dark:border-slate-800/80 rounded-xl p-4 text-center">
            <span className="text-xs text-slate-400 dark:text-slate-500 font-normal">
              معامله‌ای در این مرحله نیست
            </span>
          </div>
        ) : (
          stageDeals.map((deal) => (
            <DealCard
              key={deal.id}
              deal={deal}
              onClick={() => onSelectDeal?.(deal.id)}
            />
          ))
        )}
      </div>
    </div>
  );
};

export default KanbanColumn;

/**
 * BidarCRM - useKanbanDeals Hook
 *
 * Coordinates multi-pipeline deal queries, stage aggregations, and optimistic
 * drag-and-drop state transitions with automatic rollback on server errors.
 */

import { useState, useEffect, useMemo } from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';
import { apiClient } from '../../../lib/api-client';
import { queryKeys } from '../../../lib/query-keys';
import { Opportunity, Pipeline, PipelineStage, EspoListResponse } from '../../../types/espocrm';

export const DEFAULT_PIPELINE_STAGES: PipelineStage[] = [
  { id: 'stage-1', name: 'سرنخ جدید', order: 1, color: '#6366f1', mappedStatus: 'Prospecting' },
  { id: 'stage-2', name: 'مذاکره اولیه', order: 2, color: '#3b82f6', mappedStatus: 'Qualification' },
  { id: 'stage-3', name: 'پیش‌نویس قرارداد', order: 3, color: '#10b981', mappedStatus: 'Proposal' },
];

export interface UseKanbanDealsReturn {
  deals: Opportunity[];
  stages: PipelineStage[];
  pipelines: Pipeline[];
  selectedPipelineId: string;
  setSelectedPipelineId: (id: string) => void;
  isLoading: boolean;
  isError: boolean;
  error: string | null;
  moveDeal: (
    dealId: string,
    targetStageId: string,
    targetStageName: string,
    lossReason?: string,
    lossReasonNote?: string
  ) => Promise<void>;
  refetch: () => Promise<void>;
  clearError: () => void;
}

export function useKanbanDeals(initialPipelineId = ''): UseKanbanDealsReturn {
  const queryClient = useQueryClient();
  const [selectedPipelineId, setSelectedPipelineId] = useState<string>(initialPipelineId);
  const [localDeals, setLocalDeals] = useState<Opportunity[]>([]);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  // 1. Fetch Pipelines
  const pipelineQuery = useQuery<EspoListResponse<Pipeline>>({
    queryKey: queryKeys.pipelines.all,
    queryFn: () => apiClient.get<EspoListResponse<Pipeline>>('/Pipeline'),
    staleTime: 1000 * 60 * 5,
  });

  const pipelines = pipelineQuery.data?.list || [];
  const activePipelineId = selectedPipelineId || pipelines[0]?.id || 'default';
  const activePipeline = pipelines.find((p) => p.id === activePipelineId) || pipelines[0];

  // Fallback query for stages if pipeline response does not contain inline stages
  const stagesQuery = useQuery<EspoListResponse<PipelineStage>>({
    queryKey: ['pipelineStages', activePipelineId],
    queryFn: async () => {
      try {
        const endpoint = activePipelineId && activePipelineId !== 'default'
          ? `/PipelineStage?maxSize=100&orderBy=order&where[0][type]=equals&where[0][attribute]=pipelineId&where[0][value]=${encodeURIComponent(activePipelineId)}`
          : '/PipelineStage?maxSize=100&orderBy=order';
        return await apiClient.get<EspoListResponse<PipelineStage>>(endpoint);
      } catch {
        return { list: [], total: 0 };
      }
    },
    staleTime: 1000 * 60 * 5,
  });

  const stages = useMemo<PipelineStage[]>(() => {
    if (activePipeline?.stages && activePipeline.stages.length > 0) {
      return activePipeline.stages;
    }
    if (stagesQuery.data?.list && stagesQuery.data.list.length > 0) {
      return stagesQuery.data.list;
    }
    return DEFAULT_PIPELINE_STAGES;
  }, [activePipeline, stagesQuery.data]);

  // 2. Fetch Opportunities (EspoCRM Opportunity does not store pipelineId in DB, so query cleanly)
  const dealsQuery = useQuery<EspoListResponse<Opportunity>>({
    queryKey: queryKeys.deals.kanban(activePipelineId),
    queryFn: () => apiClient.get<EspoListResponse<Opportunity>>('/Opportunity?maxSize=200'),
    staleTime: 1000 * 60 * 2,
  });

  // Sync server deals to local state, filtering by pipelineId if set
  useEffect(() => {
    if (dealsQuery.data?.list) {
      const list = activePipelineId && activePipelineId !== 'default'
        ? dealsQuery.data.list.filter((d) => !d.pipelineId || d.pipelineId === activePipelineId)
        : dealsQuery.data.list;
      setLocalDeals(list);
    }
  }, [dealsQuery.data, activePipelineId]);

  // 3. Optimistic Move Action with Rollback & Invalidation
  const moveDeal = async (
    dealId: string,
    targetStageId: string,
    targetStageName: string,
    lossReason?: string,
    lossReasonNote?: string
  ): Promise<void> => {
    setErrorMessage(null);
    const previousSnapshot = [...localDeals];

    const targetStage = stages.find((s) => s.id === targetStageId || s.name === targetStageName);
    const stageValue = targetStage?.mappedStatus || targetStageName;

    // Optimistically update local board state
    setLocalDeals((prev) =>
      prev.map((d) =>
        d.id === dealId
          ? {
              ...d,
              pipelineStageId: targetStageId,
              stage: stageValue,
              ...(lossReason ? { lossReason, lossReasonNote } : {}),
            }
          : d
      )
    );

    try {
      const payload: Record<string, unknown> = {};
      if (targetStageId) {
        payload.pipelineStageId = targetStageId;
      }
      if (stageValue) {
        payload.stage = stageValue;
      }
      if (lossReason) {
        payload.lossReason = lossReason;
      }
      if (lossReasonNote) {
        payload.lossReasonNote = lossReasonNote;
      }

      await apiClient.patch(`/Opportunity/${dealId}`, payload);
      queryClient.invalidateQueries({ queryKey: queryKeys.deals.all });
      queryClient.invalidateQueries({ queryKey: queryKeys.dashboard.all });
    } catch (err: unknown) {
      // Rollback on failure
      setLocalDeals(previousSnapshot);
      setErrorMessage('خطا در انتقال معامله. تغییرات بازگردانده شد.');
      throw err;
    }
  };

  const refetch = async () => {
    const [, dRes] = await Promise.all([pipelineQuery.refetch(), dealsQuery.refetch()]);
    if (dRes.data?.list) {
      const list =
        activePipelineId && activePipelineId !== 'default'
          ? dRes.data.list.filter((d) => !d.pipelineId || d.pipelineId === activePipelineId)
          : dRes.data.list;
      setLocalDeals(list);
    }
  };

  const clearError = () => {
    setErrorMessage(null);
  };

  return {
    deals: localDeals,
    stages,
    pipelines,
    selectedPipelineId: activePipelineId,
    setSelectedPipelineId,
    isLoading: (dealsQuery.isLoading && localDeals.length === 0) || (pipelineQuery.isLoading && pipelines.length === 0),
    isError: dealsQuery.isError || pipelineQuery.isError,
    error: errorMessage || (dealsQuery.error instanceof Error ? dealsQuery.error.message : null),
    moveDeal,
    refetch,
    clearError,
  };
}

export default useKanbanDeals;

/**

 * Core layout frame with Didar-inspired dark signature header & rail,

 */

import React, { useState, useEffect, useRef } from 'react';
import { Menu, Pin, X, Briefcase, User, Building2 } from 'lucide-react';
import { Sidebar } from './Sidebar';
import { GlobalSearchBar } from './GlobalSearchBar';
import { UserNav } from './UserNav';
import { brandNameFa } from '../../lib/brand';
import { PinnedItem, getPinnedItems, removePinnedItem } from '../../lib/pinned';
import { toPersianDigits } from '../../lib/iran-format';

export interface AppShellProps {
  children?: React.ReactNode;
  currentPath?: string;
  onNavigate?: (path: string) => void;
  onNewDeal?: () => void;
  onSearch?: (query: string) => void;
  className?: string;
}

export const AppShell: React.FC<AppShellProps> = ({
  children,
  currentPath = '/dashboard',
  onNavigate,
  onNewDeal,
  onSearch: _onSearch,
  className = '',
}) => {
  const [mobileMenuOpen, setMobileMenuOpen] = useState(false);
  const [showPinnedDropdown, setShowPinnedDropdown] = useState(false);
  const [pinnedItems, setPinnedItems] = useState<PinnedItem[]>([]);
  const pinnedDropdownRef = useRef<HTMLDivElement>(null);

  // Sync pinned items
  useEffect(() => {
    setPinnedItems(getPinnedItems());

    const handleUpdate = (e: Event) => {
      const customEvent = e as CustomEvent<{ items: PinnedItem[] }>;
      if (customEvent.detail?.items) {
        setPinnedItems(customEvent.detail.items);
      } else {
        setPinnedItems(getPinnedItems());
      }
    };

    window.addEventListener('bidar:pinned_updated', handleUpdate);
    return () => window.removeEventListener('bidar:pinned_updated', handleUpdate);
  }, []);

  // Click outside to close pinned dropdown
  useEffect(() => {
    const handleClickOutside = (e: MouseEvent) => {
      if (pinnedDropdownRef.current && !pinnedDropdownRef.current.contains(e.target as Node)) {
        setShowPinnedDropdown(false);
      }
    };
    if (showPinnedDropdown) {
      document.addEventListener('mousedown', handleClickOutside);
    }
    return () => document.removeEventListener('mousedown', handleClickOutside);
  }, [showPinnedDropdown]);

  const handlePinnedClick = (item: PinnedItem) => {
    setShowPinnedDropdown(false);
    onNavigate?.(item.url);
  };

  const handleUnpin = (e: React.MouseEvent, id: string) => {
    e.stopPropagation();
    removePinnedItem(id);
    setPinnedItems((prev) => prev.filter((i) => i.id !== id));
  };

  const getPinnedIcon = (type: string) => {
    switch (type) {
      case 'deal':
        return <Briefcase className="w-3.5 h-3.5 text-indigo-400" />;
      case 'contact':
        return <User className="w-3.5 h-3.5 text-emerald-400" />;
      case 'account':
        return <Building2 className="w-3.5 h-3.5 text-amber-400" />;
      default:
        return <Pin className="w-3.5 h-3.5 text-slate-400" />;
    }
  };

  return (
    <div
      className={`bidar-app-shell flex h-screen w-screen overflow-hidden bg-slate-900 text-slate-100 font-sans antialiased selection:bg-emerald-500 selection:text-white ${className}`}
      dir="rtl"
      lang="fa"
    >
      {/* 1. Primary RTL Sidebar (Right side - Slim Rail with Desktop Expand & Mobile Drawer) */}
      <Sidebar
        currentPath={currentPath}
        onNavigate={onNavigate}
        mobileOpen={mobileMenuOpen}
        onCloseMobile={() => setMobileMenuOpen(false)}
      />

      {/* 2. Main Work Area (Header + Fluid Responsive Canvas) */}
      <div className="flex-1 flex flex-col min-w-0 h-screen overflow-hidden">
        {/* Top Header Bar - Dark Didar Signature Topbar */}
        <header className="h-14 bg-slate-900 border-b border-slate-800/90 px-3 sm:px-6 flex items-center justify-between gap-3 z-30 flex-shrink-0">
          <div className="flex items-center gap-2 sm:gap-4 flex-1 max-w-xl">
            {/* Hamburger Button on Mobile/Tablet */}
            <button
              type="button"
              onClick={() => setMobileMenuOpen(true)}
              className="p-1.5 text-slate-400 hover:text-slate-100 hover:bg-slate-800 rounded-lg lg:hidden cursor-pointer"
              aria-label="باز کردن منو"
            >
              <Menu className="w-5 h-5" />
            </button>

            {/* Workspace Identifier */}
            <div className="hidden md:flex items-center gap-2 pl-2 border-l border-slate-800">
              <span className="text-xs font-semibold text-slate-200">{brandNameFa} CRM</span>
              <span className="text-[10px] bg-emerald-500/15 text-emerald-400 border border-emerald-500/20 px-1.5 py-0.5 rounded-sm">
                کاریز فعال
              </span>
            </div>

            {/* Global Search Bar */}
            <GlobalSearchBar onNavigate={onNavigate} />

            {/* Pinned Items Dropdown Trigger */}
            <div className="relative hidden xl:block" ref={pinnedDropdownRef}>
              <button
                type="button"
                onClick={() => setShowPinnedDropdown(!showPinnedDropdown)}
                className={`flex items-center gap-1.5 px-2.5 py-1.5 rounded-xl text-xs transition cursor-pointer ${
                  showPinnedDropdown
                    ? 'bg-slate-800 text-emerald-400'
                    : 'text-slate-400 hover:text-slate-200 hover:bg-slate-800/60'
                }`}
                title="موارد پین‌شده"
              >
                <Pin className={`w-3.5 h-3.5 ${pinnedItems.length > 0 ? 'fill-emerald-500/30 text-emerald-400' : ''}`} />
                <span>پین‌شده‌ها</span>
                {pinnedItems.length > 0 && (
                  <span className="font-mono text-[10px] bg-slate-800 text-slate-300 px-1.5 py-0.2 rounded-full">
                    {toPersianDigits(pinnedItems.length)}
                  </span>
                )}
              </button>

              {/* Pinned Items Popover Menu */}
              {showPinnedDropdown && (
                <div className="absolute left-0 mt-2 w-72 bg-white dark:bg-slate-900 border border-slate-200 dark:border-slate-800 rounded-2xl shadow-2xl z-50 overflow-hidden animate-in fade-in duration-100">
                  <div className="p-3 border-b border-slate-100 dark:border-slate-800 flex items-center justify-between text-xs font-bold text-slate-900 dark:text-slate-100">
                    <span>موارد سنجاق‌شده ({toPersianDigits(pinnedItems.length)})</span>
                    <span className="text-[10px] font-normal text-slate-400">حداکثر ۱۰ مورد</span>
                  </div>

                  <div className="max-h-72 overflow-y-auto divide-y divide-slate-100 dark:divide-slate-800/60 text-xs">
                    {pinnedItems.length === 0 ? (
                      <div className="p-6 text-center text-slate-400 text-[11px]">
                        موردی سنجاق نشده است. با زدن علامت سنجاق در صفحات جزئیات، موارد پرکاربرد را اینجا ذخیره کنید.
                      </div>
                    ) : (
                      pinnedItems.map((item) => (
                        <div
                          key={item.id}
                          onClick={() => handlePinnedClick(item)}
                          className="flex items-center justify-between p-2.5 hover:bg-slate-50 dark:hover:bg-slate-800/60 transition cursor-pointer group"
                        >
                          <div className="flex items-center gap-2 min-w-0">
                            {getPinnedIcon(item.type)}
                            <div className="min-w-0">
                              <span className="font-medium text-slate-900 dark:text-slate-100 truncate block">
                                {item.title}
                              </span>
                              {item.subtitle && (
                                <span className="text-[10px] text-slate-400 truncate block">
                                  {item.subtitle}
                                </span>
                              )}
                            </div>
                          </div>
                          <button
                            type="button"
                            onClick={(e) => handleUnpin(e, item.id)}
                            className="p-1 text-slate-400 hover:text-rose-500 rounded opacity-0 group-hover:opacity-100 transition cursor-pointer"
                            title="حذف از پین‌شده‌ها"
                          >
                            <X className="w-3.5 h-3.5" />
                          </button>
                        </div>
                      ))
                    )}
                  </div>
                </div>
              )}
            </div>
          </div>

          {/* User & Theme Navigation */}
          <UserNav onNewDeal={onNewDeal} onNavigate={onNavigate} />
        </header>

        {/* Dynamic Responsive Workspace Canvas (Light / Dark Adaptive) */}
        <main className="flex-1 overflow-y-auto overflow-x-hidden p-3 sm:p-5 lg:p-6 bg-[#F4F6F8] dark:bg-[#0B0F19] text-slate-800 dark:text-slate-100 transition-colors duration-150">
          {children}

};

export const API_UNAUTHORIZED_EVENT = 'bidar:unauthorized';

/**
 * Safely encodes a UTF-8 string to Base64 (supporting non-Latin and Persian characters).
 */
export function encodeBase64Utf8(str: string): string {
  const bytes = new TextEncoder().encode(str);
  let binary = '';
  const len = bytes.byteLength;
  for (let i = 0; i < len; i++) {
    binary += String.fromCharCode(bytes[i]);
  }
  return btoa(binary);
}

export class ApiError extends Error {
  public readonly status: number;
  public readonly statusText: string;
  public readonly data: unknown;

  constructor(status: number, statusText: string, data?: unknown, customMessage?: string) {
    let message = customMessage;
    if (!message) {
      let details = '';
      if (typeof data === 'object' && data !== null && 'message' in data && typeof (data as { message: unknown }).message === 'string') {
        details = (data as { message: string }).message;
      } else if (typeof data === 'string' && data.length > 0) {
        details = data;
      }

      const base = `HTTP ${status}: ${statusText || 'Request failed'}`;
      message = details ? `${base} - ${details}` : base;
    }

    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.statusText = statusText;
    this.data = data;

    // Maintain prototype chain for instanceof checks
    Object.setPrototypeOf(this, ApiError.prototype);
  }
}

export interface RequestOptions extends Omit<RequestInit, 'body'> {
  params?: Record<string, string | number | boolean | null | undefined>;
  body?: unknown;

export class ApiClient {
  private baseUrl: string;
  private auth: { username: string; token: string } | null = null;
  private unauthorizedListeners = new Set<(error: ApiError) => void>();

  constructor(baseUrl = '/api/v1') {
    this.baseUrl = baseUrl.replace(/\/+$/, '');
  }

  /**
   * Configures active authentication credentials for EspoCRM token-based API access.
   */
  public setAuth(username: string, token: string): void {
    this.auth = { username, token };
  }

  /**
   * Clears stored credentials.
   */
  public clearAuth(): void {
    this.auth = null;
  }

  /**
   * Registers a callback listener executed whenever a 401 Unauthorized response is intercepted.
   * Returns an unsubscribe function.
   */
  public onUnauthorized(callback: (error: ApiError) => void): () => void {
    this.unauthorizedListeners.add(callback);
    return () => {
      this.unauthorizedListeners.delete(callback);
    };
  }

  /**
   * Executes a typed HTTP request against EspoCRM REST API.
   */
  public async request<T = unknown>(endpoint: string, options: RequestOptions = {}): Promise<T> {
    const { params, body, headers: customHeaders, method = 'GET', ...fetchInit } = options;

    // 1. Build destination URL
    let fullUrl = endpoint.startsWith('http://') || endpoint.startsWith('https://')
      ? endpoint
      : `${this.baseUrl}/${endpoint.replace(/^\/+/, '')}`;

    if (params) {
      const searchParams = new URLSearchParams();
      for (const [key, value] of Object.entries(params)) {
        if (value !== undefined && value !== null) {
          searchParams.append(key, String(value));
        }
      }
      const queryString = searchParams.toString();
      if (queryString) {
        fullUrl += (fullUrl.includes('?') ? '&' : '?') + queryString;
      }
    }

    // 2. Prepare headers
    const headers = new Headers(customHeaders);
    if (!headers.has('Accept')) {
      headers.set('Accept', 'application/json');
    }

    if (this.auth?.username && this.auth?.token) {
      const tokenHeader = encodeBase64Utf8(`${this.auth.username}:${this.auth.token}`);
      headers.set('Espo-Authorization', tokenHeader);
      headers.set('Espo-Authorization-By-Token', 'true');
    }

    let payloadBody: BodyInit | null | undefined = undefined;
    if (body !== undefined && body !== null) {
      if (typeof body === 'string' || body instanceof FormData || body instanceof Blob) {
        payloadBody = body as BodyInit;
      } else {
        if (!headers.has('Content-Type')) {
          headers.set('Content-Type', 'application/json');
        }
        payloadBody = JSON.stringify(body);
      }
    }

    // 3. Perform network call with mandatory credentials: 'include'
    const response = await fetch(fullUrl, {
      ...fetchInit,
      method,
      headers,
      body: payloadBody,
      credentials: 'include',
    });

    // 4. Handle 204 No Content
    if (response.status === 204 || response.headers.get('content-length') === '0') {
      if (!response.ok) {
        throw new ApiError(response.status, response.statusText);
      }
      return null as unknown as T;
    }

    // 5. Parse response body safely
    const contentType = response.headers.get('content-type') || '';
    let responseData: unknown = null;

    if (contentType.includes('application/json')) {
      try {
        responseData = await response.json();
      } catch {
        responseData = null;
      }
    } else {
      try {
        responseData = await response.text();
      } catch {
        responseData = null;
      }
    }

    // 6. Handle HTTP Error statuses
    if (!response.ok) {
      const error = new ApiError(response.status, response.statusText, responseData);

      if (response.status === 401) {
        // Broadcast custom browser event
        if (typeof window !== 'undefined') {
          window.dispatchEvent(
            new CustomEvent(API_UNAUTHORIZED_EVENT, { detail: { error } })
          );
        }

        // Notify registered subscribers
        for (const listener of this.unauthorizedListeners) {
          try {
            listener(error);
          } catch (listenerError) {
            console.error('Error in onUnauthorized listener:', listenerError);
          }
        }
      }

      throw error;
    }

    return responseData as T;
  }

  public get<T = unknown>(endpoint: string, options?: RequestOptions): Promise<T> {
    return this.request<T>(endpoint, { ...options, method: 'GET' });
  }

  public post<T = unknown>(endpoint: string, body?: unknown, options?: RequestOptions): Promise<T> {
    return this.request<T>(endpoint, { ...options, method: 'POST', body });
  }

  public put<T = unknown>(endpoint: string, body?: unknown, options?: RequestOptions): Promise<T> {
    return this.request<T>(endpoint, { ...options, method: 'PUT', body });
  }

  public patch<T = unknown>(endpoint: string, body?: unknown, options?: RequestOptions): Promise<T> {
    return this.request<T>(endpoint, { ...options, method: 'PATCH', body });
  }

  public delete<T = unknown>(endpoint: string, options?: RequestOptions): Promise<T> {
    return this.request<T>(endpoint, { ...options, method: 'DELETE' });
  }

  /**
   * Downloads a binary Blob (e.g. CSV, Excel, PDF) using authenticated headers.
   */
  public async getBlob(endpoint: string, options: Omit<RequestOptions, 'body'> = {}): Promise<Blob> {
    const { params, headers: customHeaders, ...fetchInit } = options;
    let fullUrl = endpoint.startsWith('http://') || endpoint.startsWith('https://')
      ? endpoint
      : `${this.baseUrl}/${endpoint.replace(/^\/+/, '')}`;

    if (params) {
      const searchParams = new URLSearchParams();
      for (const [key, value] of Object.entries(params)) {
        if (value !== undefined && value !== null) {
          searchParams.append(key, String(value));
        }
      }
      const queryString = searchParams.toString();
      if (queryString) {
        fullUrl += (fullUrl.includes('?') ? '&' : '?') + queryString;
      }
    }

    const headers = new Headers(customHeaders);
    if (this.auth?.username && this.auth?.token) {
      const tokenHeader = encodeBase64Utf8(`${this.auth.username}:${this.auth.token}`);
      headers.set('Espo-Authorization', tokenHeader);
      headers.set('Espo-Authorization-By-Token', 'true');
    }

    const response = await fetch(fullUrl, {
      ...fetchInit,
      method: 'GET',
      headers,
      credentials: 'include',
    });

    if (!response.ok) {
      throw new ApiError(response.status, response.statusText);
    }

    return await response.blob();
  }
}

export function formatRelativeJalali(isoOrDate: string | Date | null | undefined): string {
  if (!isoOrDate) return '';
  const d = typeof isoOrDate === 'string' ? new Date(isoOrDate) : isoOrDate;
  if (!(d instanceof Date) || isNaN(d.getTime())) return '';

/**
 * Determines whether a given Jalali (Shamsi) year is a leap year (366 days, Esfand has 30 days).
 * Uses the canonical Birashk 33-year cycle algorithm matching Morilog/Jalali CalendarUtils.
 */
export function isJalaliLeapYear(jy: number): boolean {
  const breaks = [
    -61, 9, 38, 199, 426, 686, 756, 818, 1111, 1181, 1210, 1635, 2060, 2097, 2192, 2262, 2324, 2394, 2456, 3178,
  ];
  let jp = breaks[0]!;
  let jump = 0;
  for (let i = 1; i < breaks.length; i++) {
    const jm = breaks[i]!;
    jump = jm - jp;
    if (jy < jm) break;
    jp = jm;
  }
  let n = jy - jp;
  if (jump - n < 6) {
    n = n - jump + Math.floor((jump + 4) / 33) * 33;
  }
  let leap = (((n + 1) % 33) - 1) % 4;
  if (leap === -1) leap = 4;
  return leap === 0;
}

export function jalaliToGregorian(jy: number, jm: number, jd: number): [number, number, number] {
  let gy: number;
  if (jy > 979) {
    gy = 1600;
    jy -= 979;
  } else {
    gy = 621;
  }
  let days =
    365 * jy +
    Math.floor(jy / 33) * 8 +
    Math.floor(((jy % 33) + 3) / 4) +
    78 +
    jd +
    (jm < 7 ? (jm - 1) * 31 : (jm - 7) * 30 + 186);

  gy += 400 * Math.floor(days / 146097);
  days %= 146097;
  if (days > 36524) {
    gy += 100 * Math.floor(--days / 36524);
    days %= 36524;
    if (days >= 365) days++;
  }
  gy += 4 * Math.floor(days / 1461);
  days %= 1461;
  if (days > 365) {
    gy += Math.floor((days - 1) / 365);
    days = (days - 1) % 365;
  }
  let gd = days + 1;
  const sal_a = [
    0,
    31,
    (gy % 4 === 0 && gy % 100 !== 0) || gy % 400 === 0 ? 29 : 28,
    31,
    30,
    31,
    30,
    31,
    31,
    30,
    31,
    30,
    31,
  ];
  let gm = 0;
  for (gm = 0; gm < 13 && gd > sal_a[gm]!; gm++) {
    gd -= sal_a[gm]!;
  }
  return [gy, gm, gd];
}

/**
 * Checks if a given Jalali date is an Iranian official holiday or Friday.
 */
export function isIranHoliday(
  year: number,
  month: number,
  day: number
): { isHoliday: boolean; title?: string } {
  // Check solar holiday by month/day key
  const key = `${month}/${day}`;
  if (IRAN_FIXED_HOLIDAYS[key]) {
    return { isHoliday: true, title: IRAN_FIXED_HOLIDAYS[key] };
  }

  // Check if it's Friday (جمعه)
  const [gy, gm, gd] = jalaliToGregorian(year, month, day);
  const gDate = new Date(Date.UTC(gy, gm - 1, gd));
  if (gDate.getUTCDay() === 5) {
    // UTC Friday
    return { isHoliday: true, title: 'جمعه (تعطیل هفتگی)' };
  }

  return { isHoliday: false };
}

/**
 * Parses natural Persian relative expressions into Date objects.
 * Supports: "امروز", "فردا", "پس‌فردا", "هفته آینده", "ساعت ۱۰", "ساعت ۱۴:۳۰"
 */
export function parseNaturalPersianDate(input: string): Date | null {
  if (!input || typeof input !== 'string') return null;
  const raw = toAsciiDigits(input).trim().toLowerCase();

  const now = new Date();
  let target = new Date(now);

  if (raw.includes('امروز')) {
    // keep today
  } else if (raw.includes('فردا')) {
    target.setDate(target.getDate() + 1);
  } else if (raw.includes('پس‌فردا') || raw.includes('پس فردا')) {
    target.setDate(target.getDate() + 2);
  } else if (raw.includes('هفته بعد') || raw.includes('هفته آینده')) {
    target.setDate(target.getDate() + 7);
  } else {
    // Fallback: try parsing standard ISO/date
    const parsed = new Date(input);
    if (!isNaN(parsed.getTime())) return parsed;
    return null;
  }

  // Parse hour if mentioned, e.g. "ساعت 10" or "ساعت 14:30"
  const timeMatch = raw.match(/ساعت\s*(\d{1,2})(?::(\d{1,2}))?/);
  if (timeMatch && timeMatch[1]) {
    const hours = parseInt(timeMatch[1], 10);
    const minutes = timeMatch[2] ? parseInt(timeMatch[2], 10) : 0;
    target.setHours(hours, minutes, 0, 0);
  } else {
    target.setHours(10, 0, 0, 0); // Default to 10:00 AM
  }

  return target;
}

/**
 * BidarCRM - Theme Management Engine
 *
 * Supports Light and Dark modes with localStorage persistence,
 * system preference fallbacks, and instantaneous class synchronization.
 */

import { useState, useEffect, useCallback } from 'react';

export type ThemeMode = 'light' | 'dark';

export const THEME_STORAGE_KEY = 'bidar_theme';

export function getSystemTheme(): ThemeMode {
  if (typeof window !== 'undefined' && window.matchMedia) {
    return window.matchMedia('(prefers-color-scheme: dark)').matches ? 'dark' : 'light';
  }
  return 'light';
}

export function getInitialTheme(): ThemeMode {
  if (typeof window === 'undefined') return 'light';
  try {
    const stored = localStorage.getItem(THEME_STORAGE_KEY);
    if (stored === 'light' || stored === 'dark') {
      return stored;
    }
  } catch {
    // Local storage access error fallback
  }
  return getSystemTheme();
}

export function applyTheme(theme: ThemeMode): void {
  if (typeof document === 'undefined') return;
  const root = document.documentElement;
  if (theme === 'dark') {
    root.classList.add('dark');
    root.classList.remove('light');
  } else {
    root.classList.remove('dark');
    root.classList.add('light');
  }
  try {
    localStorage.setItem(THEME_STORAGE_KEY, theme);
  } catch {
    // Ignore storage quota errors
  }
}

export function toggleThemeMode(current: ThemeMode): ThemeMode {
  return current === 'dark' ? 'light' : 'dark';
}

/**
 * React Hook for theme access and toggling
 */
export function useTheme() {
  const [theme, setThemeState] = useState<ThemeMode>(getInitialTheme);

  useEffect(() => {
    applyTheme(theme);
  }, [theme]);

  // Listen to OS system preference changes if user hasn't explicitly set preference
  useEffect(() => {
    if (typeof window === 'undefined' || !window.matchMedia) return;
    const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
    const handleChange = (e: MediaQueryListEvent) => {
      const stored = localStorage.getItem(THEME_STORAGE_KEY);
      if (!stored) {
        setThemeState(e.matches ? 'dark' : 'light');
      }
    };
    mediaQuery.addEventListener('change', handleChange);
    return () => mediaQuery.removeEventListener('change', handleChange);
  }, []);

  const toggle = useCallback(() => {
    setThemeState((prev) => {
      const next = toggleThemeMode(prev);
      applyTheme(next);
      return next;
    });
  }, []);

  const setTheme = useCallback((newTheme: ThemeMode) => {
    setThemeState(newTheme);
    applyTheme(newTheme);
  }, []);

  return {
    theme,
    isDark: theme === 'dark',
    toggleTheme: toggle,
    setTheme,
  };
}
